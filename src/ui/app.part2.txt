function proj(x,y,z,w,h,model){const r=project.room,spaceW=Math.max(r.width,model.width+300),spaceD=Math.max(r.depth,model.depth+500),spaceH=r.height;if(view==='front'){const s=Math.min((w-80)/spaceW,(h-130)/spaceH)*zoom;return[40+x*s,h-70-y*s]}if(view==='top'){const s=Math.min((w-80)/spaceW,(h-110)/spaceD)*zoom;return[40+x*s,55+z*s]}const c=Math.cos(yaw),sn=Math.sin(yaw),X=(x-spaceW/2)*c-(z-spaceD/2)*sn,Z=(x-spaceW/2)*sn+(z-spaceD/2)*c,s=Math.min(w/(spaceW+spaceD*.75),h/(spaceH+spaceD*.42))*1.25*zoom;return[w/2+X*s,h*.73-y*s-Z*s*.34]}
