function poly(points,fill,stroke='#7f8c90',alpha=1){ctx.beginPath();points.forEach((p,i)=>i?ctx.lineTo(...p):ctx.moveTo(...p));ctx.closePath();ctx.globalAlpha=alpha;ctx.fillStyle=fill;ctx.fill();ctx.globalAlpha=1;ctx.strokeStyle=stroke;ctx.lineWidth=.8;ctx.stroke()}
function drawRoom(w,h,model){const r=project.room,p=(x,y,z)=>proj(x,y,z,w,h,model);poly([p(0,0,0),p(r.width,0,0),p(r.width,0,r.depth),p(0,0,r.depth)],r.floorColor,'#738086',.96);poly([p(0,0,0),p(r.width,0,0),p(r.width,r.height,0),p(0,r.height,0)],r.wallColor,'#879497',.93);poly([p(0,0,0),p(0,0,r.depth),p(0,r.height,r.depth),p(0,r.height,0)],r.wallColor,'#879497',.86);ctx.strokeStyle='#83909555';ctx.lineWidth=.7;for(let x=500;x<r.width;x+=500){ctx.beginPath();ctx.moveTo(...p(x,0,0));ctx.lineTo(...p(x,0,r.depth));ctx.stroke()}for(let z=500;z<r.depth;z+=500){ctx.beginPath();ctx.moveTo(...p(0,0,z));ctx.lineTo(...p(r.width,0,z));ctx.stroke()}if(project.ui.showDimensions){dim(p(0,r.height+70,0),p(r.width,r.height+70,0),`${r.width} ${t('mm')}`);dim(p(r.width+80,0,0),p(r.width+80,0,r.depth),`${r.depth} ${t('mm')}`);dim(p(0,0,0),p(0,r.height,0),`${r.height} ${t('mm')}`)}}
function dim(a,b,label){ctx.save();ctx.strokeStyle='#24363b';ctx.fillStyle='#1b2d32';ctx.lineWidth=1;ctx.beginPath();ctx.moveTo(...a);ctx.lineTo(...b);ctx.stroke();const x=(a[0]+b[0])/2,y=(a[1]+b[1])/2;ctx.font='700 11px system-ui';const tw=ctx.measureText(label).width+12;ctx.fillStyle='#ffffffea';ctx.fillRect(x-tw/2,y-10,tw,18);ctx.fillStyle='#17262b';ctx.textAlign='center';ctx.textBaseline='middle';ctx.fillText(label,x,y-1);ctx.restore()}
function shade(hex,amt){const n=parseInt(hex.slice(1),16),r=Math.max(0,Math.min(255,(n>>16)+amt)),g=Math.max(0,Math.min(255,((n>>8)&255)+amt)),b=Math.max(0,Math.min(255,(n&255)+amt));return`rgb(${r},${g},${b})`}
function drawCircleObject(o,w,h,model){const c=proj(...o.center,w,h,model),sx=Math.max(8,o.size[0]*.055*zoom),sy=Math.max(8,o.size[1]*.055*zoom);ctx.beginPath();ctx.ellipse(c[0],c[1],sx,sy,0,0,Math.PI*2);ctx.fillStyle=o.kind==='appliance-glass'?'#15232a':'#909a9d';ctx.fill();ctx.strokeStyle='#c5ccce';ctx.lineWidth=2;ctx.stroke();return true}
